% --- Page geometry ----------------------------------------------------
% The ICLR style sets a 5.5 in text block inside 8.5 in paper, i.e. 1.5 in of
% margin on EACH side. That is a conference-proceedings convention, not a
% requirement for an internship report, and here it was simply wasting paper:
% widening the text block to 6.4 in and the height to 9.2 in fits noticeably
% more text on every page and removes several pages of blank margin, without
% changing a single word. The side margin is now (8.5 - 6.4)/2 = 1.05 in, so
% \oddsidemargin = 1.05 - 1 = 0.05 in (LaTeX measures from a 1 in origin).
\setlength{\textwidth}{6.4in}
\setlength{\oddsidemargin}{0.05in}
\setlength{\evensidemargin}{0.05in}
\setlength{\textheight}{9.2in}

% --- Float placement -------------------------------------------------
% This report is table-heavy, and LaTeX's defaults refuse to put more than
% 70% of a page into floats. That was leaving half-empty pages all over the
% appendix. Loosening the four fractions below lets a big table share a page
% with the text around it instead of pushing everything down.
\renewcommand{\topfraction}{0.92}
\renewcommand{\bottomfraction}{0.85}
\renewcommand{\textfraction}{0.06}
\renewcommand{\floatpagefraction}{0.80}

% --- Whitespace around floats, captions and equations -----------------
% The defaults leave large gaps above and below every figure and table, and
% this report has a lot of both. Tightening them removes several pages of
% blank space without making anything harder to read.
\setlength{\textfloatsep}{10pt plus 2pt minus 3pt}   % float next to top/bottom of a page
\setlength{\intextsep}{9pt plus 2pt minus 3pt}       % float in the middle of the text
\setlength{\floatsep}{9pt plus 2pt minus 3pt}        % between two floats
\setlength{\abovecaptionskip}{5pt}
\setlength{\belowcaptionskip}{0pt}

% Captions one size down and justified, so a long caption costs fewer lines.
\captionsetup{font=small, labelfont=bf, skip=5pt}
\captionsetup[sub]{font=small, labelfont=bf}

% Tighter lists. This report has many itemize/enumerate blocks (design goals,
% limitations, future work), and the default vertical space around and inside a
% list adds up to about a page over the whole document.
\usepackage{enumitem}
\setlist{topsep=3pt, partopsep=0pt, parsep=0pt, itemsep=2pt, leftmargin=1.6em}

% Usage:  \figph{Overall SEA-Net architecture: input -> encoder -> pooling head}

\newcommand{\figph}[2][0.92]{%
  \setlength{\fboxsep}{0pt}%
  \fcolorbox{black!45}{black!6}{%
    \parbox[c][0.30\linewidth][c]{#1\linewidth}{%
      \centering\ttfamily\footnotesize FIGURE PLACEHOLDER\\[4pt]%
      \normalfont\itshape\small #2\\[4pt]%
      \ttfamily\scriptsize replace this line with \textbackslash includegraphics}}}
